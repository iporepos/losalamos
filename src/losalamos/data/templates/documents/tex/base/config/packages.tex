% ==============================================================
% config/packages.tex
% Core package configuration for DocumentTeX base template.
%
% Isolation strategy
% ------------------
% The following concerns are intentionally excluded from this
% file and expected to be supplied by variant or user overlay:
%
%   config/fonts.tex          — fontenc, inputenc, font packages
%   config/language.tex       — babel, datetime2, csquotes, nomencl,
%                               cleveref, hyperref
%   config/bibliography.tex   — biblatex style, backend, options
%
% Private/project-specific inputs (gitignored):
%
%   locals/bibliography.tex   — \addbibresource declarations
%
% Load order contract
% -------------------
%   fonts → language → (everything else) → bibliography
%
% cleveref and hyperref are loaded at the end of language.tex
% to guarantee:  babel → cleveref → hyperref
%
% ==============================================================

% --------------------------------------------------------------
% Fonts (overlayable)
\input{config/fonts}

% --------------------------------------------------------------
