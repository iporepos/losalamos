% Glossary entries

\newglossaryentry{curve-number}
{
    name=curve number,
    description={Empirical parameter used in the SCS/NRCS method to estimate direct runoff from rainfall. It encodes the hydrological response of a catchment as a function of land use, soil type, and antecedent moisture condition, ranging from 0 (no runoff) to 100 (complete runoff)}
}

\newglossaryentry{rationalism}
{
    name=rationalism,
    description={Philosophical tradition holding that deductive and innate reasoning is the primary source of knowledge and the ultimate criterion of truth. It stands in opposition to empiricism, which grounds knowledge in sensory experience}
}
