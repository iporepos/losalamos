% ==============================================================
% boxes.tex
% Styled tcolorbox environments.
% Colors driven by AccentColor defined in design/colors.tex.
% ==============================================================

\tcbuselibrary{skins, breakable}

% --------------------------------------------------------------
% Shared style preset
\tcbset{
    boxbasestyle/.style={
        enhanced,
        breakable,
        colback      = AccentColorVeryLight,
        colbacktitle = AccentColorLight,
        coltitle     = AccentColor,
        colframe     = AccentColor,
        boxrule      = 0.5pt,
        arc          = 2pt,
        fonttitle    = \bfseries\sffamily\small,
        fontupper    = \sffamily\footnotesize,
        boxsep       = 0mm,  % base padding on all sides (default 4pt)
        top          = 2mm,  % top inner padding
        bottom       = 2mm,  % bottom inner padding
        left         = 2mm,  % left inner padding
        right        = 2mm,  % right inner padding
        toptitle     = 1mm,  % top padding inside title bar
        bottomtitle  = 1mm,  % bottom padding inside title bar
    }
}

% --------------------------------------------------------------
% boxstory — numbered, breakable, document-wide counter
% Usage:
%   \begin{boxstory}{Title}
%     Content here.
%   \end{boxstory}
%
%   Optional first arg accepts any tcolorbox key, e.g.:
%   \begin{boxstory}[list text=Short entry]{Long title}
\newtcolorbox[
    auto counter,
    list inside = lob,
]{boxstory}[2][]{
    boxbasestyle,
    title = {\sffamily Box~\thetcbcounter\quad #2},
    #1,
}

% --------------------------------------------------------------
% List of Boxes command
% Usage in main.tex:
%   \tcblistof[\section*]{lob}{\trListOfBoxes}
\newcommand{\listofboxes}{%
    \tcblistof[\section*]{lob}{\trListOfBoxes}%
}
