% NOTE — informational, neutral
% Usage: \begin{amdnote} ... \end{amdnote}

    boxbasestyle,
    colback      = cyan!5,
    colbacktitle = cyan!50!black,
    colframe     = cyan!50!black,
    coltitle     = white,
    title        = {\faInfoCircle\quad \trAdmNote},
}
