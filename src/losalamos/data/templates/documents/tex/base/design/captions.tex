% --------------------------------------------------------------
% Captions
\DeclareCaptionLabelSeparator{pipe}{\;|\;}

% --------------------------------------------------------------
% Figures
\captionsetup[figure]{
    font={footnotesize, sf},
    labelfont={bf, color=AccentColor},
    labelsep=pipe, % quad, colon, pipe
    width=\textwidth
}

% aligned on top for side caption
\sidecaptionvpos{figure}{t}

% --------------------------------------------------------------
% Tables
\captionsetup[table]{
    font={footnotesize, sf},
    labelfont={bf, color=AccentColor},
    labelsep=pipe,
    width=\textwidth
}

% --------------------------------------------------------------
% Listings
% Hook into the caption package so lstlisting captions
% follow the same style as figures and tables.
\DeclareCaptionStyle{lstlisting}{
    font={footnotesize, sf},
    labelfont={bf, color=AccentColor},
    labelsep=pipe,
    width=\textwidth,
    justification=centering
}
\captionsetup[lstlisting]{style=lstlisting}
