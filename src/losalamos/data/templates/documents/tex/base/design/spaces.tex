% --------------------------------------------------------------
% Page size and margins
\geometry{
 a4paper,
 left=40mm,
 right=20mm,
 top=10mm,
 bottom=25mm
}

% --------------------------------------------------------------
% Custom spaces
\newcommand{\pagesectionvspace}{\vspace*{15mm}}
% Text body tab space
\newcommand{\bodytab}{10mm}

% --------------------------------------------------------------
% Paragraph indentation space
\setlength{\parindent}{5mm}

% --------------------------------------------------------------
% Paragraph spacing
\setlength{\parskip}{2mm} 

% --------------------------------------------------------------
% Space between float and surrounding text
\setlength{\intextsep}{3mm}      % above and below in-text floats (h)
\setlength{\textfloatsep}{2mm}   % above/below top (t) and bottom (b) floats
\setlength{\floatsep}{2mm}       % between two consecutive floats

% --------------------------------------------------------------
% Space between float and its caption
\setlength{\abovecaptionskip}{1mm}
\setlength{\belowcaptionskip}{1mm}

% --------------------------------------------------------------
% Line spacing
\linespread{1}

% --------------------------------------------------------------
% Section spacing
\titlespacing*{\section}      {0mm}{6mm}{2mm}
\titlespacing*{\subsection}   {0mm}{2mm}{0mm}
\titlespacing*{\subsubsection}{0mm}{2mm}{0mm}

% --------------------------------------------------------------
% Header spacing
\setlength{\headheight}{15mm}
\setlength{\headsep}{5mm}

% --------------------------------------------------------------
% Footer spacing
\setlength{\footskip}{10mm}
