% ==============================================================
% titles.tex
% Section title formatting.
% Heading color is driven by HeadingColor defined in design/colors.tex.
% ==============================================================

\titleformat{\section}
  {\bfseries\sffamily\Large\color{HeadingColor}}
  {\thesection}{5mm}{}

\titleformat{\subsection}
  {\bfseries\sffamily\large\color{HeadingColor}}
  {\thesubsection}{4mm}{}

\titleformat{\subsubsection}
  {\sffamily\normalsize\color{HeadingColor}}
  {\thesubsubsection}{4mm}{}

\titleformat{\paragraph}[runin]
  {\bfseries\sffamily\normalsize\color{HeadingColor}}
  {}{0mm}{}[.]
